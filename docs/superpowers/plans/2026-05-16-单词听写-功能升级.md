```javascript
// miniprogram/utils/compare.js
function normalize(s) {
  const trimmed = (s || '').trim()
  // 只对全英文做 lowercase；含中文时保留原样
  return /^[a-zA-Z]+$/.test(trimmed) ? trimmed.toLowerCase() : trimmed
}

function compareWords(original, written) {
  return original.map((word, i) => {
    const w = written[i] || ''
    return {
      original: word,
      written: w || '（未识别）',
      correct: normalize(word) === normalize(w)
    }
  })
}

module.exports = { compareWords, normalize }
```

```javascript
// miniprogram/utils/tts.js
function isChinese(word) {
  return /[一-龥]/.test(word)
}

function buildUrl(word) {
  // 有道 dictvoice：type=2 英文，type=1 中文（实测中文支持有限，
  // 失败时调用方应当跳过而不阻塞）
  const type = isChinese(word) ? 1 : 2
  return `https://dict.youdao.com/dictvoice?audio=${encodeURIComponent(word)}&type=${type}`
}

function playWord(word) {
  return new Promise((resolve, reject) => {
    const audio = wx.createInnerAudioContext()
    audio.src = buildUrl(word)
    let done = false
    const finish = (err) => {
      if (done) return
      done = true
      audio.destroy()
      err ? reject(err) : resolve()
    }
    audio.onEnded(() => finish())
    audio.onError((e) => finish(e))
    audio.play()
  })
}

function sleep(seconds) {
  return new Promise(resolve => setTimeout(resolve, seconds * 1000))
}

module.exports = { playWord, sleep, isChinese }
```

```javascript
// miniprogram/utils/ocr.js
// 封装微信 OCR 插件调用。实施时根据具体插件 API 调整方法名/参数。
const ocrPlugin = requirePlugin('ocr-plugin')

/**
 * 调用微信 OCR 插件识别图片中的文字
 * @param {string} filePath 本地图片路径
 * @returns {Promise<string[]>} 切分后的词数组
 */
async function recognizeWords(filePath) {
  // 不同插件 API 不同。常见模式：
  // 1) 通用文字识别 API，返回整段文本
  // 2) 我们再按空白切词
  const res = await ocrPlugin.generalOcr({ imgPath: filePath })
  const fullText = res.text || (res.items || []).map(i => i.text).join('\n')
  return splitToWords(fullText)
}

/**
 * 把识别到的文本按空白/换行切分成词数组
 * - 中文按整个字符串切分（暂不分词）
 * - 英文按空白切分
 */
function splitToWords(text) {
  if (!text) return []
  return text
    .split(/[\s\n\t,，、;；]+/)
    .map(w => w.trim())
    .filter(w => w.length > 0)
}

/**
 * 入口函数：让用户拍照/选图，识别后返回词数组
 */
function chooseAndRecognize() {
  return new Promise((resolve, reject) => {
    wx.chooseMedia({
      count: 1,
      mediaType: ['image'],
      sourceType: ['album', 'camera'],
      success: async (res) => {
        try {
          const words = await recognizeWords(res.tempFiles[0].tempFilePath)
          resolve(words)
        } catch (e) {
          reject(e)
        }
      },
      fail: reject
    })
  })
}

module.exports = { recognizeWords, chooseAndRecognize, splitToWords }
```

```javascript
const { getSessions } = require('../../utils/db')

Page({
  data: {
    sessions: [],
    loading: true
  },

  onShow() {
    this.loadSessions()
  },

  async loadSessions() {
    this.setData({ loading: true })
    try {
      const res = await getSessions(50)
      const sessions = (res.data || []).map(this.enrichSession)
      this.setData({ sessions, loading: false })
    } catch (e) {
      console.error('加载单词本失败', e)
      this.setData({ loading: false })
    }
  },

  enrichSession(s) {
    return {
      ...s,
      dateStr: formatDate(s.createdAt),
      scoreColor: s.score === s.total ? '#34C759'
                : s.score / s.total >= 0.6 ? '#FF9500'
                : '#FF3B30'
    }
  }
})

function formatDate(d) {
  const date = new Date(d)
  const now = new Date()
  const isToday = date.toDateString() === now.toDateString()
  const yesterday = new Date(now); yesterday.setDate(now.getDate() - 1)
  const isYesterday = date.toDateString() === yesterday.toDateString()
  const hh = String(date.getHours()).padStart(2,'0')
  const mm = String(date.getMinutes()).padStart(2,'0')
  if (isToday) return `今天 ${hh}:${mm}`
  if (isYesterday) return `昨天 ${hh}:${mm}`
  return `${date.getMonth()+1}/${date.getDate()} ${hh}:${mm}`
}
```

```javascript
const { playWord, sleep } = require('../../utils/tts')

Page({
  data: {
    words: [],
    playCount: 2,
    interval: 3,
    currentIndex: 0,
    playingCount: 1,
    playedCount: 0,
    dotsArray: [],
    isPlaying: false,
    paused: false,
    countdown: 0,
    finished: false,
    showWord: false,
    hiddenMask: '●●●●'
  },

  _stopped: false,
  _countdownTimer: null,

  onLoad(options) {
    const words = JSON.parse(decodeURIComponent(options.words || '[]'))
    const playCount = parseInt(options.playCount || 2)
    const interval = parseInt(options.interval || 3)
    this.setData({
      words, playCount, interval,
      dotsArray: Array(playCount).fill(0)
    })
    this.updateMask(0)
    this.startFlow()
  },

  onUnload() {
    this._stopped = true
    if (this._countdownTimer) clearInterval(this._countdownTimer)
  },

  toggleShow() {
    this.setData({ showWord: !this.data.showWord })
  },

  updateMask(index) {
    const w = this.data.words[index] || ''
    // 隐藏态：用 ● 占位，数量与字符数相同（中文按字符算）
    const len = Math.max(2, Math.min(6, w.length))
    this.setData({ hiddenMask: '●'.repeat(len) })
  },

  async startFlow() {
    const { words, playCount, interval } = this.data
    for (let i = 0; i < words.length; i++) {
      if (this._stopped) return
      this.setData({ currentIndex: i, playedCount: 0, playingCount: 1, countdown: 0 })
      this.updateMask(i)
      for (let p = 0; p < playCount; p++) {
        if (this._stopped) return
        while (this.data.paused) await sleep(0.3)
        this.setData({ isPlaying: true, playingCount: p + 1 })
        try {
          await playWord(words[i])
        } catch (e) {
          console.error('TTS error', e)
        }
        this.setData({ isPlaying: false, playedCount: p + 1 })
        if (p < playCount - 1) await this.runCountdown(interval)
      }
      if (i < words.length - 1) await sleep(1.5)
    }
    if (!this._stopped) {
      this.setData({ finished: true })
      setTimeout(() => {
        if (!this._stopped) {
          wx.redirectTo({
            url: `/pages/grading/index?words=${encodeURIComponent(JSON.stringify(words))}`
          })
        }
      }, 1500)
    }
  },

  runCountdown(seconds) {
    return new Promise(resolve => {
      this.setData({ countdown: seconds })
      this._countdownTimer = setInterval(() => {
        if (this._stopped) { clearInterval(this._countdownTimer); resolve(); return }
        if (this.data.paused) return
        const next = this.data.countdown - 1
        if (next <= 0) {
          clearInterval(this._countdownTimer)
          this.setData({ countdown: 0 })
          resolve()
        } else {
          this.setData({ countdown: next })
        }
      }, 1000)
    })
  },

  togglePause() { this.setData({ paused: !this.data.paused }) }
})
```

```javascript
const { chooseAndRecognize, recognizeWords } = require('../../utils/ocr')
const { compareWords } = require('../../utils/compare')
const { saveSession } = require('../../utils/db')

Page({
  data: {
    state: 'upload',  // 'upload' | 'loading' | 'result'
    words: [],
    results: [],
    correctCount: 0,
    scorePercent: 0,
    playCount: 2,
    interval: 3
  },

  onLoad(options) {
    const words = JSON.parse(decodeURIComponent(options.words || '[]'))
    this.setData({ words })
  },

  uploadAnswer() {
    wx.chooseMedia({
      count: 1,
      mediaType: ['image'],
      sourceType: ['album', 'camera'],
      success: async (res) => {
        this.setData({ state: 'loading' })
        try {
          const recognized = await recognizeWords(res.tempFiles[0].tempFilePath)
          this.processResult(recognized)
        } catch (e) {
          console.error('OCR error', e)
          wx.showToast({ title: '识别失败，请重试', icon: 'none' })
          this.setData({ state: 'upload' })
        }
      }
    })
  },

  processResult(writtenWords) {
    const { words } = this.data
    const results = compareWords(words, writtenWords)
    const correctCount = results.filter(r => r.correct).length
    const scorePercent = Math.round((correctCount / words.length) * 100)
    this.setData({ results, correctCount, scorePercent, state: 'result' })
    saveSession({
      words, result: results, score: correctCount, total: words.length,
      settings: { playCount: this.data.playCount, interval: this.data.interval }
    }).catch(e => console.error('保存记录失败', e))
  },

  finish() {
    wx.redirectTo({ url: '/pages/wordbook/index' })
  }
})
```
