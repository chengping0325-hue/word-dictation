const { db, _ } = require('./cloudbase')

const sessionsCol = db.collection('sessions')

async function saveSession(session) {
  const app = getApp()
  const openid = app.globalData.openid
  return sessionsCol.add({
    data: {
      openid,
      words: session.words,
      result: session.result,
      score: session.score,
      total: session.total,
      createdAt: db.serverDate()
    }
  })
}

async function getSessions(limit = 20) {
  const app = getApp()
  const openid = app.globalData.openid
  return sessionsCol
    .where({ openid })
    .orderBy('createdAt', 'desc')
    .limit(limit)
    .get()
}

module.exports = { saveSession, getSessions }

- [ ] **Step 5: 创建 JS**

`miniprogram/pages/dictation/index.js`:
```js
Page({
  data: {
    words: [],
    inputWord: '',
    playCount: 2,
    interval: 3
  },

  onInput(e) {
    this.setData({ inputWord: e.detail.value })
  },

  addWord() {
    const word = this.data.inputWord.trim()
    if (!word) return
    const words = [...this.data.words, word]
    this.setData({ words, inputWord: '' })
  },

  deleteWord(e) {
    const index = e.currentTarget.dataset.index
    const words = this.data.words.filter((_, i) => i !== index)
    this.setData({ words })
  },

  clearAll() {
    wx.showModal({
      title: '确认清空',
      content: '确定要清空所有单词吗？',
      success: (res) => {
        if (res.confirm) this.setData({ words: [] })
      }
    })
  },

  decPlayCount() {
    if (this.data.playCount <= 1) return
    this.setData({ playCount: this.data.playCount - 1 })
  },

  incPlayCount() {
    if (this.data.playCount >= 5) return
    this.setData({ playCount: this.data.playCount + 1 })
  },

  decInterval() {
    if (this.data.interval <= 1) return
    this.setData({ interval: this.data.interval - 1 })
  },

  incInterval() {
    if (this.data.interval >= 10) return
    this.setData({ interval: this.data.interval + 1 })
  },

  photoOCR() {
    wx.showToast({ title: '功能开发中', icon: 'none' })
    // Task 5 实现
  },

  startDictation() {
    if (this.data.words.length === 0) return
    wx.navigateTo({
      url: `/pages/playing/index?words=${JSON.stringify(this.data.words)}&playCount=${this.data.playCount}&interval=${this.data.interval}`
    })
  }
})
```

// 使用有道词典音频接口播放英文单词
function playWord(word) {
  return new Promise((resolve, reject) => {
    const audio = wx.createInnerAudioContext()
    // type=2 为美式发音
    audio.src = `https://dict.youdao.com/dictvoice?audio=${encodeURIComponent(word)}&type=2`
    audio.onEnded(() => {
      audio.destroy()
      resolve()
    })
    audio.onError((err) => {
      audio.destroy()
      reject(err)
    })
    audio.play()
  })
}

function sleep(seconds) {
  return new Promise(resolve => setTimeout(resolve, seconds * 1000))
}

module.exports = { playWord, sleep }

`miniprogram/pages/playing/index.js`:
```js
const { playWord, sleep } = require('../../utils/tts')

Page({
  data: {
    words: [],
    playCount: 2,
    interval: 3,
    currentIndex: 0,
    playingCount: 1,
    playedCount: 0,
    dotsArray: [],
    isPlaying: false,
    paused: false,
    countdown: 0
  },

  _stopped: false,
  _countdownTimer: null,

  onLoad(options) {
    const words = JSON.parse(options.words || '[]')
    const playCount = parseInt(options.playCount || 2)
    const interval = parseInt(options.interval || 3)
    this.setData({
      words,
      playCount,
      interval,
      dotsArray: Array(playCount).fill(0)
    })
    this.startFlow()
  },

  onUnload() {
    this._stopped = true
    if (this._countdownTimer) clearInterval(this._countdownTimer)
  },

  async startFlow() {
    const { words, playCount, interval } = this.data
    for (let i = 0; i < words.length; i++) {
      if (this._stopped) return
      this.setData({ currentIndex: i, playedCount: 0, playingCount: 1 })
      for (let p = 0; p < playCount; p++) {
        if (this._stopped) return
        while (this.data.paused) await sleep(0.3)
        this.setData({ isPlaying: true, playingCount: p + 1 })
        await playWord(words[i])
        this.setData({ isPlaying: false, playedCount: p + 1 })
        if (p < playCount - 1) await sleep(interval)
      }
      if (i < words.length - 1) {
        await this.runCountdown(interval)
      }
    }
    // 全部播放完毕，跳转判分页
    if (!this._stopped) {
      setTimeout(() => {
        wx.redirectTo({
          url: `/pages/grading/index?words=${JSON.stringify(words)}`
        })
      }, 1000)
    }
  },

  runCountdown(seconds) {
    return new Promise(resolve => {
      this.setData({ countdown: seconds })
      this._countdownTimer = setInterval(() => {
        if (this._stopped) { clearInterval(this._countdownTimer); resolve(); return }
        const next = this.data.countdown - 1
        if (next <= 0) {
          clearInterval(this._countdownTimer)
          this.setData({ countdown: 0 })
          resolve()
        } else {
          this.setData({ countdown: next })
        }
      }, 1000)
    })
  },

  togglePause() {
    this.setData({ paused: !this.data.paused })
  },

  goBack() {
    this._stopped = true
    wx.navigateBack()
  }
})
```

const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })

const tencentcloud = require('tencentcloud-sdk-nodejs-ocr')
const OcrClient = tencentcloud.ocr.v20181119.Client

const SECRET_ID = 'YOUR_SECRET_ID'   // 替换为真实值
const SECRET_KEY = 'YOUR_SECRET_KEY' // 替换为真实值

exports.main = async (event) => {
  const { fileID } = event

  // 从 CloudBase Storage 获取图片 base64
  const res = await cloud.downloadFile({ fileID })
  const base64 = res.fileContent.toString('base64')

  const client = new OcrClient({
    credential: { secretId: SECRET_ID, secretKey: SECRET_KEY },
    region: 'ap-shanghai'
  })

  const result = await client.GeneralBasicOCR({
    ImageBase64: base64,
    LanguageType: 'ENG'
  })

  // 提取识别出的单词（每行一个单词）
  const words = result.TextDetections
    .map(d => d.DetectedText.trim().toLowerCase())
    .filter(w => /^[a-z]+$/.test(w))

  return { words }
}

const { saveSession } = require('../../utils/db')
const { compareWords } = require('../../utils/compare')

Page({
  data: {
    words: [],
    results: [],
    correctCount: 0,
    scorePercent: 0,
    hasResult: false,
    loading: false
  },

  onLoad(options) {
    const words = JSON.parse(options.words || '[]')
    this.setData({ words })
  },

  uploadAnswer() {
    wx.chooseMedia({
      count: 1,
      mediaType: ['image'],
      sourceType: ['album', 'camera'],
      success: (res) => {
        const tempFilePath = res.tempFiles[0].tempFilePath
        this.runOCR(tempFilePath)
      }
    })
  },

  async runOCR(filePath) {
    this.setData({ loading: true })
    try {
      // 上传图片到 CloudBase Storage
      const uploadRes = await wx.cloud.uploadFile({
        cloudPath: `ocr/${Date.now()}.jpg`,
        filePath
      })
      // 调用云函数 OCR
      const ocrRes = await wx.cloud.callFunction({
        name: 'ocrWords',
        data: { fileID: uploadRes.fileID }
      })
      const writtenWords = ocrRes.result.words
      this.processResult(writtenWords)
    } catch (e) {
      wx.showToast({ title: 'OCR 识别失败，请重试', icon: 'none' })
    } finally {
      this.setData({ loading: false })
    }
  },

  processResult(writtenWords) {
    const { words } = this.data
    const results = compareWords(words, writtenWords)
    const correctCount = results.filter(r => r.correct).length
    const scorePercent = Math.round((correctCount / words.length) * 100)
    this.setData({ results, correctCount, scorePercent, hasResult: true })
    // 保存到数据库
    saveSession({ words, result: results, score: correctCount, total: words.length })
  },

  finish() {
    wx.switchTab
    ? wx.navigateBack({ delta: 2 })
    : wx.navigateTo({ url: '/pages/profile/index' })
  }
})

- [ ] **Step 8: 创建 compare.js**

`miniprogram/utils/compare.js`:
```js
/**
 * 将原始单词列表和识别出的单词列表逐一对比
 * 策略：按顺序匹配（one-to-one），大小写不敏感
 */
function compareWords(original, written) {
  return original.map((word, i) => {
    const w = (written[i] || '').toLowerCase().trim()
    const o = word.toLowerCase().trim()
    return {
      original: word,
      written: written[i] || '（未识别）',
      correct: w === o
    }
  })
}

module.exports = { compareWords }
```
